      case 164: /* cmd ::= insert_cmd INTO xfullname idlist_opt VALUES LP exprlist RP SEMI */
{
    InsertStmt* insert = sqlite3MallocZero(sizeof(InsertStmt));

    // 表名应该是 yymsp[-6].minor.yy563->zName，而不是 yymsp[-8].minor.yy502（yymsp[-8].minor.yy502 是 INSERT token）
    insert->table_name = sqlite3DbStrDup(pParse->db, yymsp[-6].minor.yy563->a[0].zName);

    // 列名
    if (yymsp[-5].minor.yy204) {
        insert->column_count = yymsp[-5].minor.yy204->nId;
        insert->column_names = sqlite3MallocZero(sizeof(char*) * yymsp[-5].minor.yy204->nId);
        for (int i = 0; i < yymsp[-5].minor.yy204->nId; ++i) {
            insert->column_names[i] = sqlite3DbStrDup(pParse->db, yymsp[-5].minor.yy204->a[i].zName);
        }
    } else {
        insert->column_count = yymsp[-2].minor.yy402->nExpr; // 如果没写列名，就默认所有列
        insert->column_names = NULL;
    }

    // 值表达式
    if (yymsp[-2].minor.yy402) {
        insert->values = sqlite3MallocZero(sizeof(MiniExprList));
        insert->values->count = yymsp[-2].minor.yy402->nExpr;
        insert->values->items = sqlite3MallocZero(sizeof(MiniExpr*) * yymsp[-2].minor.yy402->nExpr);
        for (int i = 0; i < yymsp[-2].minor.yy402->nExpr; ++i) {
            insert->values->items[i] = convertExpr(yymsp[-2].minor.yy402->a[i].pExpr);
        }
    }

    // 封装成 SQLStatement
    SQLStatement* stmt = sqlite3MallocZero(sizeof(SQLStatement));
    stmt->type = STMT_INSERT;
    stmt->insert_stmt = insert;
    pParse->pMiniPGStatement = stmt;
}
